## Para pegar en el chat de Figma Make

## CONTEXTO

Agregar la funcionalidad de exportar un reporte individual por empleado
desde la sección de Reportes. El reporte debe mostrar el detalle completo
de horas trabajadas, registros y total a pagar para el período seleccionado.

Por defecto el período es el mes actual, pero el admin puede filtrar
por rango de fechas personalizado.

**No modificar:** la vista general de Reportes existente, el fichaje,
registros, empleados ni ninguna otra pantalla.

---

## CAMBIO 3 — Query de datos para el reporte individual

```js
async function getDatosReporteEmpleado(empleadoId, desde, hasta) {
  // Datos del empleado
  const { data: empleado } = await supabase
    .from('empleados')
    .select('nombre_completo, valor_hora, jornada_horas')
    .eq('id', empleadoId)
    .single()

  // Registros del período
  const query = supabase
    .from('registros')
    .select('fecha, hora_entrada, hora_salida, horas_totales, sueldo_dia, editado_manual, sospechoso')
    .eq('empleado_id', empleadoId)
    .gte('fecha', desde)
    .lte('fecha', hasta)
    .order('fecha', { ascending: true })

  const { data: registros } = await query

  // Calcular totales
  const registrosCompletos = registros.filter(r => r.hora_salida !== null)
  const totalHoras = registrosCompletos.reduce((acc, r) => acc + parseFloat(r.horas_totales || 0), 0)
  const totalPagar = registrosCompletos.reduce((acc, r) => acc + parseFloat(r.sueldo_dia || 0), 0)
  const diasTrabajados = registrosCompletos.length

  return { empleado, registros, totalHoras, totalPagar, diasTrabajados }
}
```

## CAMBIO 4 — Exportación PDF individual

Usar `jsPDF` + `jspdf-autotable` (ya instalados).

```js
function exportarPDFEmpleado(datos, desde, hasta) {
  const { empleado, registros, totalHoras, totalPagar, diasTrabajados } = datos
  const doc = new jsPDF()

  // ── HEADER naranja ────────────────────────────────
  doc.setFillColor(249, 115, 22)
  doc.rect(0, 0, 210, 35, 'F')

  doc.setTextColor(255, 255, 255)
  doc.setFontSize(20)
  doc.setFont('helvetica', 'bold')
  doc.text('Panahon', 14, 15)

  doc.setFontSize(10)
  doc.setFont('helvetica', 'normal')
  doc.text('Reporte individual de empleado', 14, 23)

  // ── DATOS DEL EMPLEADO ────────────────────────────
  doc.setTextColor(26, 26, 26)
  doc.setFontSize(16)
  doc.setFont('helvetica', 'bold')
  doc.text(empleado.nombre_completo, 14, 48)

  doc.setFontSize(10)
  doc.setFont('helvetica', 'normal')
  doc.setTextColor(107, 114, 128)
  doc.text(`Período: ${desde} al ${hasta}`, 14, 56)
  doc.text(`Valor hora: $${parseFloat(empleado.valor_hora).toLocaleString('es-AR')}`, 14, 62)
  doc.text(`Generado el: ${new Date().toLocaleDateString('es-AR')}`, 14, 68)

  // ── RESUMEN ───────────────────────────────────────
  doc.setFillColor(255, 244, 236)
  doc.rect(14, 74, 182, 22, 'F')

  doc.setTextColor(249, 115, 22)
  doc.setFontSize(10)
  doc.setFont('helvetica', 'bold')
  doc.text(`${diasTrabajados} días trabajados`, 20, 83)
  doc.text(`${parseFloat(totalHoras).toFixed(2)} hs totales`, 80, 83)
  doc.text(`$${parseFloat(totalPagar).toLocaleString('es-AR')} total`, 145, 83)

  doc.setTextColor(107, 114, 128)
  doc.setFontSize(8)
  doc.setFont('helvetica', 'normal')
  doc.text('Días trabajados', 20, 90)
  doc.text('Horas totales', 80, 90)
  doc.text('Total a pagar', 145, 90)

  // ── TABLA DE REGISTROS ────────────────────────────
  const filas = registros
    .filter(r => r.hora_salida !== null)
    .map(r => [
      new Date(r.fecha + 'T12:00:00').toLocaleDateString('es-AR', { weekday: 'short', day: '2-digit', month: '2-digit' }),
      new Date(r.hora_entrada).toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit', hour12: false }),
      new Date(r.hora_salida).toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit', hour12: false }),
      parseFloat(r.horas_totales).toFixed(2) + ' hs',
      '$' + parseFloat(r.sueldo_dia).toLocaleString('es-AR'),
      r.editado_manual ? 'Editado' : r.sospechoso ? '⚠️' : '✓'
    ])

  autoTable(doc, {
    startY: 102,
    head: [['Fecha', 'Entrada', 'Salida', 'Horas', 'Importe', 'Estado']],
    body: filas,
    foot: [[
      `Total: ${diasTrabajados} días`,
      '', '',
      parseFloat(totalHoras).toFixed(2) + ' hs',
      '$' + parseFloat(totalPagar).toLocaleString('es-AR'),
      ''
    ]],
    headStyles: { fillColor: [249, 115, 22], textColor: [255, 255, 255], fontStyle: 'bold' },
    footStyles: { fillColor: [244, 245, 247], fontStyle: 'bold' },
    alternateRowStyles: { fillColor: [248, 249, 250] },
    columnStyles: {
      0: { cellWidth: 35 },
      1: { halign: 'center' },
      2: { halign: 'center' },
      3: { halign: 'right' },
      4: { halign: 'right' },
      5: { halign: 'center', cellWidth: 20 }
    },
    margin: { left: 14, right: 14 }
  })

  // ── FOOTER ────────────────────────────────────────
  const pageCount = doc.getNumberOfPages()
  for (let i = 1; i <= pageCount; i++) {
    doc.setPage(i)
    doc.setFontSize(8)
    doc.setTextColor(156, 163, 175)
    doc.text(
      `Panahon — Reporte de ${empleado.nombre_completo} — Página ${i} de ${pageCount}`,
      14,
      doc.internal.pageSize.height - 10
    )
  }

  doc.save(`Panahon_${empleado.nombre_completo.replace(' ', '_')}_${desde}_${hasta}.pdf`)
}
```

## CAMBIO 5 — Exportación Excel individual

```js
function exportarExcelEmpleado(datos, desde, hasta) {
  const { empleado, registros, totalHoras, totalPagar, diasTrabajados } = datos
  const BOM = '\uFEFF'

  const encabezado = [
    `Reporte individual — ${empleado.nombre_completo}`,
    `Período: ${desde} al ${hasta}`,
    `Valor hora: $${parseFloat(empleado.valor_hora).toLocaleString('es-AR')}`,
    `Generado por Panahon`,
    ``
  ].join('\n')

  const columnas = ['Fecha', 'Entrada', 'Salida', 'Horas', 'Importe', 'Estado']

  const filas = registros
    .filter(r => r.hora_salida !== null)
    .map(r => [
      new Date(r.fecha + 'T12:00:00').toLocaleDateString('es-AR'),
      new Date(r.hora_entrada).toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit', hour12: false }),
      new Date(r.hora_salida).toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit', hour12: false }),
      parseFloat(r.horas_totales).toFixed(2),
      parseFloat(r.sueldo_dia).toFixed(2),
      r.editado_manual ? 'Editado manualmente' : r.sospechoso ? 'Revisar' : 'Completo'
    ].join(','))

  const filaTotales = [
    `Total: ${diasTrabajados} días`,
    '', '',
    parseFloat(totalHoras).toFixed(2),
    parseFloat(totalPagar).toFixed(2),
    ''
  ].join(',')

  const contenido = [
    encabezado,
    columnas.join(','),
    ...filas,
    filaTotales
  ].join('\n')

  const blob = new Blob([BOM + contenido], { type: 'text/csv;charset=utf-8;' })
  const url = URL.createObjectURL(blob)
  const link = document.createElement('a')
  link.href = url
  link.download = `Panahon_${empleado.nombre_completo.replace(' ', '_')}_${desde}_${hasta}.csv`
  document.body.appendChild(link)
  link.click()
  document.body.removeChild(link)
  URL.revokeObjectURL(url)
}
```

## FLUJO COMPLETO

```
Admin en Reportes
  → Ve tabla con resumen por empleado
  → Toca [↓ PDF] o [↓ Excel] en la fila de un empleado
  → Se abre modal con período pre-cargado (mes actual)
  → Admin ajusta fechas si necesita
  → Toca "Descargar PDF" o "Descargar Excel"
  → Se descarga el archivo con el reporte individual
```
